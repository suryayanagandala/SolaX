        [Test]
        public async Task AssigendTechnician_Create_Success()
        {
            var users = new Users
            {
                UserId = "MUMBAI",
                FullName = "Abhi",
                EmailId = "adbhi@gmail.com",
                Password = BCrypt.Net.BCrypt.HashPassword("Abhi@2003"),
                Address = "Kondapalli vijaywada",
                PhoneNumber = "9494962341",
                CreatedBy = "Admin",
                CreatedDate = TimeUtility.NowIST(),
                RoleId = (int)RoleIds.Customer
            };

            _db.Users.Add(users);
            _db.SaveChanges();

            
            var dto = new TechnicianCreateDto
            {
                UserId = "MUMBAI",
                Specialization = Specialization.Installer,
                Qualification = Qualification.Diploma,
                Salary = 30000,
                Discipline = Discipline.Mechanical,
                ServicePrice = 1200,
                ExperienceYears = 1

            };

            await technicianRepository.AssignedTechnician(dto);
            var techniciansCount = await _db.Technicians.CountAsync();
            Assert.That(techniciansCount, Is.EqualTo(2));

            var technician = await _db.Technicians.FirstOrDefaultAsync(x=>x.UserId=="MUMBAI");
            Assert.That(technician!.UserId, Is.EqualTo("MUMBAI"));
            Assert.That(technician.Salary, Is.EqualTo(30000));
            Assert.That(technician.User.RoleId, Is.EqualTo(2));

        }
