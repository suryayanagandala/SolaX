using ApplicationLayer.Interfaces;
using CoreModels;
using CoreModels.DTO_S;
using CoreModels.Entities;
using CoreModels.Enums;
using Microsoft.AspNetCore.Http;
using Microsoft.EntityFrameworkCore;
using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace DbAccessaLayer.Infrastructure
{
    public class AdminRepository : IAdminRepository
    {
        private readonly ApplicationDbContext _db;

        public AdminRepository(ApplicationDbContext db)
        {
            _db = db;
        }
        public async Task<AdminDashboardStatsDto> GetStats()
        {
            return new AdminDashboardStatsDto
            {
                Users = await _db.Users.CountAsync(),

                Employees = await _db.Users.CountAsync(x => x.RoleId == (int)RoleIds.Technician),

                Orders = await _db.Orders.CountAsync(),

                Products = await _db.Products.CountAsync(),

                TotalEmployess = await _db.Technicians.Where(x => x.IsActive).CountAsync(),

                PanlesCount = await _db.Panels.Where(x => x.IsActive).CountAsync(),

                Installation = await _db.Installations.Where(x => x.Status == InstallationStatus.Completed.ToString()).CountAsync(),

                Delieveries = await _db.Deliveries.Where(x => x.Status == DeliveryStatus.Delivered.ToString()).CountAsync()
            };

        }

        public async Task<IEnumerable<Users>> GetAllUsers()
        {
            try
            {
                var customersData = await _db.Users.Where(x => x.RoleId == 3).OrderByDescending(x => x.CreatedDate).ToListAsync();
                return customersData;
            }
            catch (Exception ex)
            {
                throw new Exception(ex.Message);
            }
        }

        public async Task<object> GetGlobalProduction()
        {
            try
            {
                var currentYear = TimeUtility.CurrentYear();
                var todayDate = TimeUtility.StartOffDay();
                var currentMonth = TimeUtility.NowIST().Month;
                var previousMonth= TimeUtility.NowIST().Month-1;

                var monthlyTrend=await _db.SolarPerformances.Where(x=>x.Generated_at.Year==currentYear).GroupBy(x=>x.Generated_at.Month)
                                .Select(k=> new
                                {
                                    Month=k.Key,
                                    TotalPower=k.Sum(x=>x.Power)
                                }).OrderBy(x=>x.Month) .ToListAsync();
                var dailyTrend=await _db.SolarPerformances.Where(x=>x.Generated_at.Date==todayDate).GroupBy(x=>x.Generated_at.Hour)
                               .Select(x=> new
                               {
                                   Hour=x.Key,
                                   ToalPower=x.Sum(x=>x.Power)
                               }).OrderBy(x=>x.Hour) .ToListAsync();
                var currentMonthTotal = monthlyTrend.Where(x => x.Month == currentMonth).Select(x => x.TotalPower).FirstOrDefault();
                var previousMonthTotal=monthlyTrend.Where(x=>x.Month==previousMonth).Select(x=>x.TotalPower).FirstOrDefault();
                decimal growthPercent = 0m;

                if (previousMonthTotal.HasValue && previousMonthTotal > 0)
                {
                    growthPercent =
                        ((currentMonthTotal.GetValueOrDefault() - previousMonthTotal.Value)/ previousMonthTotal.Value) * 100m;
                }

                return new 
                {
                    monthlyTrend,
                    dailyTrend,
                    changePercent=growthPercent,
                };
            }
            catch (Exception)
            {
                throw;
            }
        }

        public async Task<object> ProductionTrend()
        {
            try
            {

                var currentDate = TimeUtility.NowIST();
                var currentMonth = TimeUtility.NowIST().Month;
                var previousMonth=TimeUtility.NowIST().Month-1;
                var currentYear= TimeUtility.NowIST().Year;

                var currentSales=await _db.OrderItems.Where(x=>x.Order.Status==OrderStatus.DELIVERED&&x.Order.OrderDate
                                .Month==currentMonth &&x.Order.OrderDate.Year==currentYear).
                                GroupBy(x=> new
                                {
                                    x.ProductId,
                                    x.Product.ProductName,
                                    x.Product.ProductImages.First().ImageUrl,
                                }).Select(x=> new
                                {
                                    x.Key.ProductName,
                                    x.Key.ProductId,
                                    x.Key.ImageUrl,
                                    TotalQuantity=x.Sum(x=>x.Quantity)
                                }).ToListAsync();
                var previousSales = await _db.OrderItems.Where(x => x.Order.Status == OrderStatus.DELIVERED && x.Order.OrderDate.Month == previousMonth &&
                                   x.Order.OrderDate.Year == currentYear).GroupBy(x => new
                                   {
                                       x.ProductId,
                                       x.Product.ProductName,
                                       x.Product.ProductImages.First().ImageUrl,
                                   }).Select(x => new
                                   {
                                       x.Key.ProductName,
                                       x.Key.ProductId,
                                       x.Key.ImageUrl,
                                       TotalQuantity = x.Sum(x => x.Quantity),
                                   }).ToListAsync();
                var result = currentSales.Select(x =>
                {
                    var previous = previousSales.FirstOrDefault(x => x.ProductId == x.ProductId);
                    double percent = 0;
                    if (previous != null && previous.TotalQuantity > 0)
                    {
                        percent = ((x.TotalQuantity - previous.TotalQuantity) / (previous.TotalQuantity)) * 100;
                    }
                    return new
                    {
                        x.ImageUrl,
                        x.ProductId,
                        x.ProductName,
                        x.TotalQuantity,
                        changePercentage = percent,
                    };
                }).OrderByDescending(x=>x.changePercentage);
                return result;
                                  
            }
            catch (Exception ex)
            {
                throw new Exception(ex.Message);
            }
        }

        public async Task<object> GetProductionUpdate()
        {
            try
            {
                var todayData = TimeUtility.NowIST().Date;
                var offlinePanles = await _db.SolarPerformances.Include(x => x.Panel).Where(x => x.PanelId != x.Panel.PanelId && x.Generated_at.Date == todayData)
                                .CountAsync();

                var lowProductionPanles = await _db.SolarPerformances.Where(x => x.IsActive).GroupBy(x => new
                {
                    x.PanelId,
                    x.Power,
                }).Select(x => new
                {
                    TotalPower = x.Sum(x => x.Power),
                    x.Key
                }).Where(x => x.TotalPower < 500).CountAsync();

                var lowProductStock=await _db.Products.Where(x=>x.Stock<=20&&x.IsActive).Select(x=> new { x.ProductId, x.ProductName,x.Stock }).ToListAsync();

                return new
                {
                    offlinePanles,
                    lowProductionPanles,
                    lowProductStock
                };
            }
            catch (Exception)
            {
                throw;
            }
        }


        public async Task<List<object>> SoldProducts()
        {
            try
            {
                var soldProducts = await _db.OrderItems.Where(x => x.Order.Status == OrderStatus.DELIVERED).
                                 GroupBy(x => new
                                 {
                                     x.ProductId,
                                     x.Product.ProductName,
                                     x.Product.ProductImages.First().ImageUrl,
                                 }).Select(x => new
                                 {
                                     x.Key.ProductId,
                                     x.Key.ProductName,
                                     x.Key.ImageUrl,
                                     Quantity = x.Sum(x => x.Quantity)
                                 }).ToListAsync();
                return new List<object>
                {
                    soldProducts
                };
            }
            catch (Exception)
            {
                throw;
            }
        }
    }
}
