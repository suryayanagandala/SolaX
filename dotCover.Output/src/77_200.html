    public static class TestDataSeeder
    {
        public static void SeedBasicOrder(ApplicationDbContext context)
        {
            try
            {
                var roles = DataInitializer.GetRoles();
                context.Roles.AddRange(roles);
                context.SaveChanges();

                var users = DataInitializer.GetUsers();
                context.Users.AddRange(users);
                context.SaveChanges();

                // ---------- CATEGORIES ----------
                var categories = DataInitializer.GetCategories();
                context.Categories.AddRange(categories);
                context.SaveChanges();

                // ---------- PRODUCTS ----------
                var products = DataInitializer.GetProducts();

                // IMPORTANT: Ensure CategoryId matches existing category
                foreach (var product in products)
                {
                    product.CategoryId = 1;
                }

                context.Products.AddRange(products);
                context.SaveChanges();

                // ---------- ORDERS ----------
                var orders = DataInitializer.GetOrders();

                foreach (var order in orders)
                {
                    order.UserId = "ERCXET";   // must exist

                    foreach (var item in order.OrderItem)
                    {
                        item.ProductId = 1;   // must exist
                    }
                }

                context.Orders.AddRange(orders);
                context.SaveChanges();

                // ---------- CART ----------
                var carts = DataInitializer.GetCarts();

                foreach (var cart in carts)
                {
                    cart.UserId = "ABX123";   // must exist

                    foreach (var item in cart.CartItems)
                    {
                        item.ProductId = 1;   // must exist
                    }
                }

                context.Carts.AddRange(carts);
                context.SaveChanges();

                // ---------- TECHNICIANS ----------
                var technicians = DataInitializer.GetTechnicians();

                foreach (var tech in technicians)
                {
                    tech.UserId = "ABX123";   // must exist
                }

                context.Technicians.AddRange(technicians);

                var payment = new Payments
                {
                    PaymentId = 1,
                    OrderId = orders[0].OrderId,
                    Order = orders[0],
                    PaymentMethod = "COD",
                    PaymentStatus = "Success",
                    TransactionId = "TXN12345",
                    Amount = orders[0].TotalAmount,
                    PaymentDate = DateTimeOffset.UtcNow, 
                    IsActive = true
                };
                context.Payments.Add(payment);
                context.SaveChanges();
            }
            catch (Exception ex)
            {
                Console.WriteLine("MAIN ERROR:");
                Console.WriteLine(ex.Message);

                if (ex.InnerException != null)
                {
                    Console.WriteLine("INNER ERROR:");
                    Console.WriteLine(ex.InnerException.Message);
                }

                throw;
            }
        }
    }
