using ApplicationLayer.Interfaces;
using CoreModels;
using CoreModels.DTO_S;
using CoreModels.Entities;
using Microsoft.EntityFrameworkCore;
using System;
using System.Collections.Generic;
using System.Linq;
using System.Runtime.InteropServices;
using System.Text;
using System.Threading.Tasks;

namespace DbAccessaLayer.Infrastructure
{
    public class RoleRepository : IRoleRepository
    {
        public ApplicationDbContext _db;
        public RoleRepository(ApplicationDbContext db)
        {
            _db = db;
        }
        public async Task<IEnumerable<Roles>> GetAllRoles()
        {
            try
            {
                var data = await _db.Roles.ToListAsync();
                return data;
            }
            catch(Exception ex)
            {
                throw new Exception(ex.Message);
            }
        }

        public List<User> GetUsersFromUser(string role)
        {
            try
            {
                var user=(from r in _db.Set<Roles>() join u in _db.Set<Users>() on r.RoleId equals u.RoleId
                          where r.RoleName==role && r.IsActive select new User
                          {
                              UserId=u.UserId,
                              RoleId=r.RoleId,
                              FullName=u.FullName,
                              EmailId=u.EmailId,
                          }).ToList();
                return user;
            }
            catch (Exception)
            {
                throw;
            }   
        }
    }
}
